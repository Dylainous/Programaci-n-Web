-- ==========================================
-- 1. CREACIÓN DE TABLAS
-- ==========================================

-- Tabla principal: Estudiantes
CREATE TABLE estudiantes (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  email VARCHAR(255) UNIQUE NOT NULL,
  fecha_nacimiento DATE NOT NULL, -- Dato importante a validar en la capa de presentación/lógica
  creado_en TIMESTAMPTZ DEFAULT now() NOT NULL
);

-- Tabla secundaria: Cursos
CREATE TABLE cursos (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  profesor VARCHAR(100) NOT NULL,
  creado_en TIMESTAMPTZ DEFAULT now() NOT NULL
);

-- Tabla intermedia (Relación Muchos a Muchos)
CREATE TABLE inscripciones (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  estudiante_id UUID NOT NULL,
  curso_id UUID NOT NULL,
  fecha_inscripcion TIMESTAMPTZ DEFAULT now() NOT NULL,

  -- ==========================================
  -- 2. DEFINICIÓN DE RELACIONES (FOREIGN KEYS)
  -- ==========================================
  CONSTRAINT fk_estudiante
    FOREIGN KEY (estudiante_id)
    REFERENCES estudiantes(id)
    ON DELETE CASCADE, -- Si se borra el estudiante, se borra su inscripción automáticamente

  CONSTRAINT fk_curso
    FOREIGN KEY (curso_id)
    REFERENCES cursos(id)
    ON DELETE CASCADE
);
